=== FILE: src/billing/pricing.py ===
"""Line pricing and invoice totals for the billing sample.

This module implements the SDD change tracked by ``SPEC-BILL-001``: the
invoice total supports a promotion code.  The behaviour that already existed
(subtotal computation, tax application and rounding) is preserved exactly;
see ``SPEC-BILL-002`` for those invariants.
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import date
from decimal import Decimal, ROUND_HALF_UP
from types import MappingProxyType
from typing import Mapping

from .money import Money, MoneyError

__all__ = [
    "ExpiredPromoError",
    "InvoiceTotals",
    "LineItem",
    "PROMO_CATALOG",
    "PricingError",
    "Promo",
    "UnknownPromoError",
    "compute_invoice_totals",
    "compute_subtotal",
]


class PricingError(ValueError):
    """Base class for pricing rule violations."""


class UnknownPromoError(PricingError):
    """Raised when a promotion code is not part of the catalog."""


class ExpiredPromoError(PricingError):
    """Raised when a promotion code is no longer valid."""


@dataclass(frozen=True)
class LineItem:
    """One priced order line."""

    sku: str
    quantity: int
    unit_price: Money

    def __post_init__(self) -> None:
        if not self.sku:
            raise PricingError("line item requires a sku")
        if not isinstance(self.quantity, int) or isinstance(self.quantity, bool):
            raise PricingError("line item quantity must be an integer")
        if self.quantity <= 0:
            raise PricingError("line item quantity must be positive")
        if not isinstance(self.unit_price, Money):
            raise PricingError("line item unit price must be a Money instance")


@dataclass(frozen=True)
class InvoiceTotals:
    """The four monetary totals reported on an invoice."""

    subtotal: Money
    discount: Money
    tax: Money
    total: Money


@dataclass(frozen=True)
class Promo:
    """One entry of the promotion code catalog."""

    percent_off: Decimal
    max_discount: Money
    expires_on: date | None = None

    def is_expired(self, today: date) -> bool:
        """Return whether the promotion stopped being valid before ``today``."""
        return self.expires_on is not None and self.expires_on < today


PROMO_CATALOG: Mapping[str, Promo] = MappingProxyType(
    {
        "SAVE10": Promo(percent_off=Decimal("10"), max_discount=Money.of("50.00")),
        "SAVE20": Promo(percent_off=Decimal("20"), max_discount=Money.of("100.00")),
        "OLD15": Promo(
            percent_off=Decimal("15"),
            max_discount=Money.of("30.00"),
            expires_on=date(2020, 6, 30),
        ),
    }
)


def compute_subtotal(items: tuple[LineItem, ...] | list[LineItem]) -> Money:
    """Return the sum of every line total, before tax and before any discount."""
    subtotal = Money.zero()
    for item in items:
        subtotal = subtotal + item.unit_price * item.quantity
    return subtotal


def _lookup_promo(promo_code: str, *, today: date) -> Promo:
    """Return the catalog entry of ``promo_code`` or raise the matching error."""
    promo = PROMO_CATALOG.get(promo_code)
    if promo is None:
        raise UnknownPromoError(f"unknown promotion code: {promo_code}")
    if promo.is_expired(today):
        raise ExpiredPromoError(f"promotion code expired: {promo_code}")
    return promo


def _compute_discount(subtotal: Money, promo: Promo) -> Money:
    """Return the discount of ``promo`` on ``subtotal``.

    The raw percentage discount is rounded to whole cents with
    ``ROUND_HALF_UP`` and then capped by the promotion's maximum discount and
    by the subtotal itself, so the discount is never negative and never
    exceeds the subtotal.
    """
    raw = subtotal.scaled(promo.percent_off / Decimal("100"))
    capped = min(raw.amount, promo.max_discount.amount, subtotal.amount)
    return Money(max(capped, Decimal("0")))


def compute_invoice_totals(
    items: tuple[LineItem, ...] | list[LineItem],
    *,
    tax_rate: Decimal,
    promo_code: str | None = None,
) -> InvoiceTotals:
    """Return the invoice totals for ``items``.

    The tax is applied to the discounted subtotal and rounded to whole cents
    with ``ROUND_HALF_UP``.  Without a promotion code the behaviour is exactly
    the one of the pre-change implementation.
    """
    if not isinstance(tax_rate, Decimal):
        raise PricingError("tax rate must be a Decimal")
    if tax_rate < Decimal("0"):
        raise PricingError("tax rate must not be negative")

    promo = _lookup_promo(promo_code, today=date.today()) if promo_code is not None else None

    subtotal = compute_subtotal(items)
    discount = _compute_discount(subtotal, promo) if promo is not None else Money.zero()

    taxable = subtotal - discount
    tax = Money((taxable.amount * tax_rate).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))
    return InvoiceTotals(
        subtotal=subtotal,
        discount=discount,
        tax=tax,
        total=taxable + tax,
    )
